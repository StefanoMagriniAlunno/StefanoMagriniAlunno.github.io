Score-PINN

The score-PINN is a neural network that takes \(\textbf{x},t\) and returns \(\nabla_x\log p_t(\textbf{x})\). We call it \(\textbf{s}_\theta\) and to train it we work on the optimization problem: - Denoising Score Matching \[ \mathbb{E}_{t\sim\mathcal{T},\textbf{x}\sim\textrm{SDE}}\left[\mathbb{E}_{\textbf{z}_\varepsilon\sim\mathcal{N}\left(\textbf{0},\varepsilon\textrm{Id}\right)}\left[\left\|\textbf{s}_\theta\left(\textbf{x} + \textbf{z}_\varepsilon,t\right) + \frac{\textbf{z}_\varepsilon}{\varepsilon}\right\|^2\right]\right] \] So to compute the loss function, we need a SDE simulator of: \[ \textrm{d}\textbf{X}_t = \textbf{f}\textrm{d}t + \sigma\textrm{d}\textbf{W}_t \] In this experiment we use Euler-Maruyama scheme: \[ \textbf{X}_{n+1} = \textbf{X}_n + \textbf{f}\Delta t + \sigma\Delta\textbf{W}_n \] where \(\Delta \textbf{W}_n\sim\mathcal{N}\left(0,\Delta t\textrm{Id}\right)\).

Score-PINN model 
import torch
import torch.nn as nn

class ScoreNetwork(nn.Module):
    def __init__(self, spatial_dim, hidden_dim):
        super().__init__()
        input_dim = spatial_dim + 1
        self.net = nn.Sequential(
            nn.Linear(input_dim, hidden_dim),
            nn.SiLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.SiLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.SiLU(),
            nn.Linear(hidden_dim, spatial_dim) 
        )

    def forward(self, x, t):
        x_t = torch.cat([x, t], dim=1)
        score = self.net(x_t)
        return score
Score matching loss function 
def score_matching_loss(score_net, x_samples, t_samples, noise_std):
    noise = torch.randn_like(x_samples)
    x_perturbed = x_samples + noise * noise_std
    predicted_score = score_net(x_perturbed, t_samples)
    target_score = -noise / noise_std
    loss = torch.mean((predicted_score - target_score)**2)
    return loss
Euler-Maruyama sampler 
def drift_function(x: torch.Tensor, t: torch.Tensor):
    # x shape (N,d)
    if x.ndim == 1:
        x = x.unsqueeze(0)
    # t shape (N,)
    if t.ndim == 0:
        t = t.unsqueeze(0)
    return -x
def sigma_function(x: torch.Tensor, t: torch.Tensor):
    # x shape (N,d)
    if x.ndim == 1:
        x = x.unsqueeze(0)
    # t shape (N,)
    if t.ndim == 0:
        t = t.unsqueeze(0)
    N, d = x.shape
    return torch.eye(d, device=x.device).repeat(N, 1, 1)

def euler_maruyama_sampler(x_init: torch.Tensor, t_init: float, t_final: float, n_steps: int):
    dt = (t_final - t_init) / n_steps

    x_samples = torch.empty((n_steps, *x_init.shape), device=x_init.device)
    t_samples = torch.linspace(t_init, t_final, n_steps, device=x_init.device)
    dW = torch.randn((n_steps, *x_init.shape), device=x_init.device) * torch.sqrt(torch.tensor(dt))

    x = x_init.detach().clone()
    for i in range(n_steps):
        x_samples[i] = x
        drift = drift_function(x, t_samples[i])
        sigma = sigma_function(x, t_samples[i])
        x = x + drift * dt + (sigma @ dW[i].unsqueeze(-1)).squeeze(-1)

    return x_samples, t_samples
